# Kattio's face for the app: the firmware's face engine (components/kattio_face/face_engine.cpp, unchanged)
cmake_minimum_required(VERSION 3.22)
project(kattio_face CXX)
set(CMAKE_CXX_STANDARD 17)
set(ENGINE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../components/kattio_face)
add_library(kattio_face SHARED face_jni.cpp ${ENGINE}/face_engine.cpp)
target_include_directories(kattio_face PRIVATE ${ENGINE})
target_compile_options(kattio_face PRIVATE -O3 -fno-exceptions -fno-rtti)
target_link_libraries(kattio_face jnigraphics log)
